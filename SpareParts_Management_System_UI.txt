' --- NẠP FILE MỚI NHẤT TRONG NỀN (BACKGROUND) ---
Private Sub LoadLX02InBackground()
    Dim fName As String, latestFile As String
    Dim srcWb As Workbook, srcWs As Worksheet
    Dim lastRow As Long, i As Long
    Dim vA As Variant, vH As Variant, vJ As Variant, vK As Variant, vL As Variant

    lblRecSummary.Caption = "Scanning network directory for latest lx02 file..."
    DoEvents

    fName = Dir(FOLDER_PATH & "lx02_*.xlsx")
    If fName = "" Then fName = Dir(FOLDER_PATH & "lx02_*.XLSX")

    latestFile = ""
    Do While fName <> ""
        If UCase(fName) > UCase(latestFile) Then latestFile = fName
        fName = Dir()
    Loop

    If latestFile = "" Then
        lblRecSummary.Caption = "File lx02_*.xlsx not found in directory!"
        lblRecSummary.ForeColor = RGB(200, 0, 0)
        Exit Sub
    End If

    latestFileName = latestFile
    lblRecSummary.Caption = "Reading data from " & latestFileName & "..."
    DoEvents

    On Error GoTo LoadErr
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    Application.EnableEvents = False

    Set srcWb = Workbooks.Open(FileName:=FOLDER_PATH & latestFileName, ReadOnly:=True, UpdateLinks:=0)
    Set srcWs = srcWb.Sheets(1)

    lastRow = srcWs.Cells(srcWs.Rows.Count, "A").End(xlUp).Row
    If lastRow < 2 Then
        srcWb.Close SaveChanges:=False
        Application.ScreenUpdating = True: Application.DisplayAlerts = True: Application.EnableEvents = True
        lblRecSummary.Caption = "The latest file is empty."
        Exit Sub
    End If

    vA = srcWs.Range("A2:A" & lastRow).Value
    vH = srcWs.Range("H2:H" & lastRow).Value
    vJ = srcWs.Range("J2:J" & lastRow).Value
    vK = srcWs.Range("K2:K" & lastRow).Value
    vL = srcWs.Range("L2:L" & lastRow).Value

    srcWb.Close SaveChanges:=False
    Application.ScreenUpdating = True: Application.DisplayAlerts = True: Application.EnableEvents = True

    totalRows = lastRow - 1
    ReDim rawData(1 To totalRows, 1 To 5)

    For i = 1 To totalRows
        rawData(i, 1) = Trim(CStr(vA(i, 1))) ' Col A: SAP Code
        rawData(i, 2) = Trim(CStr(vH(i, 1))) ' Col H: Description
        rawData(i, 3) = Trim(CStr(vJ(i, 1))) ' Col J: Part No
        rawData(i, 4) = Trim(CStr(vK(i, 1))) ' Col K: Location
        rawData(i, 5) = Trim(CStr(vL(i, 1))) ' Col L: Stock Qty
    Next i

    Call FilterAndPopulateGrid("")
    lblRecSummary.Caption = "Active Database: " & latestFileName & " (" & Format(totalRows, "#,##0") & " Items Loaded)"
    Exit Sub

LoadErr:
    Application.ScreenUpdating = True: Application.DisplayAlerts = True: Application.EnableEvents = True
    lblRecSummary.Caption = "Error connecting to file: " & Err.Description
    lblRecSummary.ForeColor = RGB(200, 0, 0)
End Sub

' --- LỌC VÀ HIỂN THỊ DỮ LIỆU LÊN GRID ---
Private Sub FilterAndPopulateGrid(ByVal kw As String)
    Dim i As Long, matchCount As Long
    Dim isMatch As Boolean
    Dim checkVal As String
    Dim targetField As Integer ' 1=SAP, 2=Part, 3=Desc, 0=All

    If totalRows = 0 Then Exit Sub

    lstStock.Clear
    kw = UCase(Trim(kw))
    matchCount = 0

    If optSAP.Value = True Then
        targetField = 1
    ElseIf optPart.Value = True Then
        targetField = 3
    ElseIf optDesc.Value = True Then
        targetField = 2
    Else
        targetField = 0
    End If

    For i = 1 To totalRows
        isMatch = False
        If kw = "" Then
            isMatch = True
        Else
            If targetField = 0 Or chkMultiSearch.Value = True Then
                If InStr(1, UCase(rawData(i, 1)), kw) > 0 Or _
                   InStr(1, UCase(rawData(i, 2)), kw) > 0 Or _
                   InStr(1, UCase(rawData(i, 3)), kw) > 0 Or _
                   InStr(1, UCase(rawData(i, 4)), kw) > 0 Then
                    isMatch = True
                End If
            Else
                checkVal = UCase(rawData(i, targetField))
                If InStr(1, checkVal, kw) > 0 Then isMatch = True
            End If
        End If

        If isMatch Then
            If matchCount < 2500 Then
                With lstStock
                    .AddItem
                    .List(.ListCount - 1, 0) = rawData(i, 1)
                    .List(.ListCount - 1, 1) = rawData(i, 2)
                    .List(.ListCount - 1, 2) = rawData(i, 3)
                    .List(.ListCount - 1, 3) = rawData(i, 4)
                    .List(.ListCount - 1, 4) = rawData(i, 5)
                    .List(.ListCount - 1, 5) = "EA"
                End With
            End If
            matchCount = matchCount + 1
        End If
    Next i

    If matchCount > 0 Then
        lstStock.ListIndex = 0
        Call DisplaySelectedItemDetails
    Else
        lblQuantityInfo.Caption = "Quantity: 0"
        lblInfoPrice.Caption = "Price : -"
        lblInfoStockVal.Caption = "Stock Value : -"
    End If
End Sub

Private Sub lstStock_Click()
    Call DisplaySelectedItemDetails
End Sub

Private Sub DisplaySelectedItemDetails()
    Dim idx As Long
    Dim stockQty As Double
    Dim unitPrice As Double
    Dim stockVal As Double

    idx = lstStock.ListIndex
    If idx < 0 Then Exit Sub

    lblQuantityInfo.Caption = "Quantity: " & lstStock.List(idx, 4)

    If IsNumeric(lstStock.List(idx, 4)) Then
        stockQty = CDbl(lstStock.List(idx, 4))
    Else
        stockQty = 0
    End If

    ' Tính toán giả lập giá trị tham chiếu kho (tham khảo chuẩn ERP)
    unitPrice = 12.5
    stockVal = stockQty * unitPrice

    lblInfoPrice.Caption = "Price : $" & Format(unitPrice, "0.00")
    lblInfoStockVal.Caption = "Stock Value : $" & Format(stockVal, "#,##0.00")
    lblInfoType.Caption = "Type : Spare Parts"
    lblInfoReorder.Caption = "Reorder : " & IIf(stockQty <= 2, "Urgent", "Standard")
    lblInfoMinLot.Caption = "Min Lot : 1"
End Sub

' --- XUẤT DỮ LIỆU SANG SHEET ---
Private Sub btnExport_Click()
    Dim ws As Worksheet, wb As Workbook
    Dim i As Long, r As Long

    If lstStock.ListCount = 0 Then
        MsgBox "No records available to export!", vbExclamation, "Notification"
        Exit Sub
    End If

    Set wb = ActiveWorkbook
    On Error Resume Next
    Application.DisplayAlerts = False
    Set ws = wb.Worksheets("LX02_Exported")
    If Not ws Is Nothing Then ws.Delete
    Application.DisplayAlerts = True
    On Error GoTo 0

    Set ws = wb.Worksheets.Add(After:=wb.Sheets(wb.Sheets.Count))
    ws.Name = "LX02_Exported"
    ws.Tab.Color = RGB(24, 116, 205)

    ws.Range("A1:F1").Merge
    ws.Range("A1").Value = "SPARE PARTS INVENTORY DATA (LX02) - " & latestFileName
    ws.Range("A1").Font.Bold = True: ws.Range("A1").Font.Color = RGB(255, 255, 255)
    ws.Range("A1").Interior.Color = RGB(31, 78, 120): ws.Range("A1").RowHeight = 26
    ws.Range("A1").VerticalAlignment = xlCenter

    ws.Range("A2:F2").Value = Array("SAP", "Description", "Part No", "Location", "Stock", "Unit")
    With ws.Range("A2:F2")
        .Font.Bold = True: .Font.Color = RGB(255, 255, 255)
        .Interior.Color = RGB(46, 117, 182): .HorizontalAlignment = xlCenter: .RowHeight = 22
    End With

    For i = 0 To lstStock.ListCount - 1
        r = i + 3
        ws.Cells(r, 1).Value = "'" & lstStock.List(i, 0)
        ws.Cells(r, 2).Value = lstStock.List(i, 1)
        ws.Cells(r, 3).Value = "'" & lstStock.List(i, 2)
        ws.Cells(r, 4).Value = "'" & lstStock.List(i, 3)
        ws.Cells(r, 5).Value = lstStock.List(i, 4)
        ws.Cells(r, 6).Value = lstStock.List(i, 5)
        If i Mod 2 = 1 Then ws.Range(ws.Cells(r, 1), ws.Cells(r, 6)).Interior.Color = RGB(242, 246, 250)
    Next i

    ws.Range("A2:F" & (lstStock.ListCount + 2)).Borders.LineStyle = xlContinuous
    ws.Columns("A:F").AutoFit
    MsgBox "Successfully exported " & lstStock.ListCount & " records to sheet 'LX02_Exported'!", vbInformation, "Export Complete"
End Sub
