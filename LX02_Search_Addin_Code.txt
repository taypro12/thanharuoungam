Private Sub LoadLatestFileInBackground()
    Dim fName As String, latestFile As String
    Dim srcWb As Workbook, srcWs As Worksheet
    Dim lastRow As Long, i As Long
    Dim vA As Variant, vH As Variant, vJ As Variant, vK As Variant, vL As Variant
    
    lblStatus.Caption = "Scanning folder for the latest file..."
    lblStatus.ForeColor = RGB(0, 102, 204)
    DoEvents
    
    fName = Dir(FOLDER_PATH & "lx02_*.xlsx")
    If fName = "" Then fName = Dir(FOLDER_PATH & "lx02_*.XLSX")
    
    latestFile = ""
    Do While fName <> ""
        If UCase(fName) > UCase(latestFile) Then latestFile = fName
        fName = Dir()
    Loop
    
    If latestFile = "" Then
        lblStatus.Caption = "Error: No 'lx02_*.xlsx' file found in folder!"
        lblStatus.ForeColor = RGB(204, 0, 0)
        lblFileInfo.Caption = "Source Folder: " & FOLDER_PATH & " (Empty)"
        MsgBox "Cannot find any file matching 'lx02_*.xlsx' in folder:" & vbCrLf & FOLDER_PATH, vbCritical, "File Not Found"
        Exit Sub
    End If
    
    latestFileName = latestFile
    lblFileInfo.Caption = "Latest File: " & latestFileName
    lblStatus.Caption = "Reading data from " & latestFileName & "..."
    DoEvents
    
    On Error GoTo ReadError
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    Application.EnableEvents = False
    
    Set srcWb = Workbooks.Open(FileName:=FOLDER_PATH & latestFileName, ReadOnly:=True, UpdateLinks:=0)
    Set srcWs = srcWb.Sheets(1)
    
    lastRow = srcWs.Cells(srcWs.Rows.Count, "A").End(xlUp).Row
    If lastRow < 2 Then
        srcWb.Close SaveChanges:=False
        Application.ScreenUpdating = True
        Application.DisplayAlerts = True
        Application.EnableEvents = True
        MsgBox "The file contains no data rows.", vbExclamation, "No Data"
        Exit Sub
    End If
    
    vA = srcWs.Range("A2:A" & lastRow).Value
    vH = srcWs.Range("H2:H" & lastRow).Value
    vJ = srcWs.Range("J2:J" & lastRow).Value
    vK = srcWs.Range("K2:K" & lastRow).Value
    vL = srcWs.Range("L2:L" & lastRow).Value
    
    srcWb.Close SaveChanges:=False
    
    Application.ScreenUpdating = True
    Application.DisplayAlerts = True
    Application.EnableEvents = True
    
    totalRecords = lastRow - 1
    ReDim allData(1 To totalRecords, 1 To 5)
    
    For i = 1 To totalRecords
        allData(i, 1) = Trim(CStr(vA(i, 1)))
        allData(i, 2) = Trim(CStr(vH(i, 1)))
        allData(i, 3) = Trim(CStr(vJ(i, 1)))
        allData(i, 4) = Trim(CStr(vK(i, 1)))
        allData(i, 5) = Trim(CStr(vL(i, 1)))
    Next i
    
    Call DisplayResults("")
    lblStatus.Caption = "Data loaded successfully (" & Format(totalRecords, "#,##0") & " rows)."
    lblStatus.ForeColor = RGB(0, 128, 0)
    Exit Sub

ReadError:
    Application.ScreenUpdating = True
    Application.DisplayAlerts = True
    Application.EnableEvents = True
    lblStatus.Caption = "Error loading file: " & Err.Description
    lblStatus.ForeColor = RGB(204, 0, 0)
    MsgBox "Failed to read file: " & Err.Description, vbCritical, "Error"
End Sub

Private Sub btnSearch_Click()
    Dim kw As String
    kw = Trim(txtKeyword.Text)
    Call DisplayResults(kw)
End Sub

Private Sub DisplayResults(ByVal kw As String)
    Dim i As Long, matchCount As Long
    Dim isMatch As Boolean
    Dim targetCol As Integer
    Dim checkVal As String
    Dim isExact As Boolean
    
    If totalRecords = 0 Then Exit Sub
    
    lstResults.Clear
    isExact = chkExactMatch.Value
    targetCol = cboSearchIn.ListIndex
    kw = UCase(kw)
    matchCount = 0
    
    For i = 1 To totalRecords
        isMatch = False
        
        If kw = "" Then
            isMatch = True
        Else
            If targetCol = 0 Then
                If isExact Then
                    If UCase(allData(i, 1)) = kw Or UCase(allData(i, 2)) = kw Or _
                       UCase(allData(i, 3)) = kw Or UCase(allData(i, 4)) = kw Or _
                       UCase(allData(i, 5)) = kw Then isMatch = True
                Else
                    If InStr(1, UCase(allData(i, 1)), kw) > 0 Or InStr(1, UCase(allData(i, 2)), kw) > 0 Or _
                       InStr(1, UCase(allData(i, 3)), kw) > 0 Or InStr(1, UCase(allData(i, 4)), kw) > 0 Or _
                       InStr(1, UCase(allData(i, 5)), kw) > 0 Then isMatch = True
                End If
            Else
                checkVal = UCase(allData(i, targetCol))
                If isExact Then
                    If checkVal = kw Then isMatch = True
                Else
                    If InStr(1, checkVal, kw) > 0 Then isMatch = True
                End If
            End If
        End If
        
        If isMatch Then
            If matchCount < 2000 Then
                With lstResults
                    .AddItem
                    .List(.ListCount - 1, 0) = matchCount + 1
                    .List(.ListCount - 1, 1) = allData(i, 1)
                    .List(.ListCount - 1, 2) = allData(i, 2)
                    .List(.ListCount - 1, 3) = allData(i, 3)
                    .List(.ListCount - 1, 4) = allData(i, 4)
                    .List(.ListCount - 1, 5) = allData(i, 5)
                End With
            End If
            matchCount = matchCount + 1
        End If
    Next i
    
    lblCount.Caption = "Found: " & Format(matchCount, "#,##0") & " record(s)"
    If matchCount = 0 Then
        lblStatus.Caption = "No records matched the search criteria."
        lblStatus.ForeColor = RGB(204, 0, 0)
    Else
        lblStatus.Caption = "Showing up to 2,000 records."
        lblStatus.ForeColor = RGB(0, 128, 0)
    End If
End Sub

Private Sub btnExport_Click()
    Dim ws As Worksheet, wb As Workbook
    Dim i As Long, r As Long
    
    If lstResults.ListCount = 0 Then
        MsgBox "No data to export!", vbExclamation, "Warning"
        Exit Sub
    End If
    
    Set wb = ActiveWorkbook
    On Error Resume Next
    Application.DisplayAlerts = False
    Set ws = wb.Worksheets("LX02_Search_Results")
    If Not ws Is Nothing Then ws.Delete
    Application.DisplayAlerts = True
    On Error GoTo 0
    
    Set ws = wb.Worksheets.Add(After:=wb.Sheets(wb.Sheets.Count))
    ws.Name = "LX02_Search_Results"
    ws.Tab.Color = RGB(0, 112, 192)
    
    ws.Range("A1:F1").Merge
    ws.Range("A1").Value = "LX02 SEARCH RESULTS - SOURCE: " & latestFileName
    ws.Range("A1").Font.Bold = True: ws.Range("A1").Font.Color = RGB(255, 255, 255)
    ws.Range("A1").Interior.Color = RGB(31, 78, 120): ws.Range("A1").RowHeight = 25
    ws.Range("A1").VerticalAlignment = xlCenter
    
    ws.Range("A2:F2").Value = Array("No.", "Column A", "Column H", "Column J", "Column K", "Column L")
    With ws.Range("A2:F2")
        .Font.Bold = True: .Font.Color = RGB(255, 255, 255)
        .Interior.Color = RGB(46, 117, 182): .HorizontalAlignment = xlCenter: .RowHeight = 22
    End With
    
    For i = 0 To lstResults.ListCount - 1
        r = i + 3
        ws.Cells(r, 1).Value = lstResults.List(i, 0)
        ws.Cells(r, 2).Value = lstResults.List(i, 1)
        ws.Cells(r, 3).Value = "'" & lstResults.List(i, 2)
        ws.Cells(r, 4).Value = "'" & lstResults.List(i, 3)
        ws.Cells(r, 5).Value = "'" & lstResults.List(i, 4)
        ws.Cells(r, 6).Value = "'" & lstResults.List(i, 5)
        
        If i Mod 2 = 1 Then
            ws.Range(ws.Cells(r, 1), ws.Cells(r, 6)).Interior.Color = RGB(242, 246, 250)
        End If
    Next i
    
    ws.Range("A2:F" & (lstResults.ListCount + 2)).Borders.LineStyle = xlContinuous
    ws.Columns("A:F").AutoFit
    MsgBox "Successfully exported " & lstResults.ListCount & " records to sheet 'LX02_Search_Results'!", vbInformation, "Success"
End Sub
