' --- SỰ KIỆN TÌM KIẾM ---
Private Sub btnSearch_Click()
    Dim kw As String
    Dim wb As Workbook, ws As Worksheet
    Dim lookAtType As XlLookAt
    Dim matchCaseVal As Boolean
    Dim totalFound As Long
    
    kw = Trim(txtKeyword.Text)
    If kw = "" Then
        MsgBox "Vui lòng nhập từ khóa cần tìm kiếm!", vbExclamation, "Chưa có từ khóa"
        txtKeyword.SetFocus
        Exit Sub
    End If
    
    Set wb = ActiveWorkbook
    If wb Is Nothing Then Exit Sub
    
    If chkExactMatch.Value = True Then lookAtType = xlWhole Else lookAtType = xlPart
    matchCaseVal = chkMatchCase.Value
    
    lstResults.Clear
    lblStatus.Caption = "Đang quét dữ liệu, vui lòng đợi giây lát..."
    lblStatus.ForeColor = RGB(0, 102, 204)
    DoEvents
    
    totalFound = 0
    
    If cboSheets.ListIndex = 0 Then
        For Each ws In wb.Worksheets
            If ws.Visible = xlSheetVisible Then
                Call ScanSheetData(ws, kw, lookAtType, matchCaseVal, totalFound)
            End If
        Next ws
    Else
        On Error Resume Next
        Set ws = wb.Worksheets(cboSheets.Value)
        On Error GoTo 0
        If Not ws Is Nothing Then Call ScanSheetData(ws, kw, lookAtType, matchCaseVal, totalFound)
    End If
    
    If totalFound = 0 Then
        lblStatus.Caption = "Không tìm thấy dữ liệu nào phù hợp!"
        lblStatus.ForeColor = RGB(204, 0, 0)
        lblCount.Caption = "Tìm thấy: 0 kết quả"
        MsgBox "Không tìm thấy ô nào chứa: """ & kw & """", vbInformation, "Kết quả"
    Else
        lblStatus.Caption = "Nhấp đúp chuột vào kết quả để di chuyển tới ô."
        lblStatus.ForeColor = RGB(0, 128, 0)
        lblCount.Caption = "Tìm thấy: " & totalFound & " kết quả"
    End If
End Sub

Private Sub ScanSheetData(ByRef ws As Worksheet, ByVal kw As String, _
                          ByVal lookVal As XlLookAt, ByVal bMatchCase As Boolean, _
                          ByRef totalFound As Long)
    Dim c As Range, firstAddr As String
    
    On Error Resume Next
    Set c = ws.Cells.Find(What:=kw, LookIn:=xlValues, LookAt:=lookVal, _
                          SearchOrder:=xlByRows, SearchDirection:=xlNext, MatchCase:=bMatchCase)
    On Error GoTo 0
    
    If Not c Is Nothing Then
        firstAddr = c.Address
        Do
            If totalFound >= 1000 Then Exit Do
            totalFound = totalFound + 1
            
            With lstResults
                .AddItem
                .List(.ListCount - 1, 0) = totalFound
                .List(.ListCount - 1, 1) = ws.Name
                .List(.ListCount - 1, 2) = c.Address(False, False)
                .List(.ListCount - 1, 3) = Left(Trim(CStr(c.Text)), 100)
                If c.HasFormula Then
                    .List(.ListCount - 1, 4) = "Công thức"
                Else
                    .List(.ListCount - 1, 4) = "Giá trị"
                End If
            End With
            
            Set c = ws.Cells.FindNext(c)
            If c Is Nothing Then Exit Do
        Loop While c.Address <> firstAddr
    End If
End Sub

' --- SỰ KIỆN NHẢY TỚI Ô (DOUBLE CLICK HOẶC BẤM NÚT) ---
Private Sub lstResults_DblClick(ByVal Cancel As MSForms.ReturnBoolean)
    Call JumpToCell
End Sub

Private Sub btnGoTo_Click()
    Call JumpToCell
End Sub

Private Sub JumpToCell()
    Dim idx As Long, shName As String, addr As String
    Dim ws As Worksheet, c As Range
    
    idx = lstResults.ListIndex
    If idx < 0 Then
        MsgBox "Vui lòng chọn một dòng kết quả trước!", vbExclamation, "Chưa chọn"
        Exit Sub
    End If
    
    shName = lstResults.List(idx, 1)
    addr = lstResults.List(idx, 2)
    
    On Error Resume Next
    Set ws = ActiveWorkbook.Worksheets(shName)
    If Not ws Is Nothing Then
        ws.Activate
        Set c = ws.Range(addr)
        c.Select
        ActiveWindow.ScrollRow = Application.WorksheetFunction.Max(1, c.Row - 4)
        ActiveWindow.ScrollColumn = Application.WorksheetFunction.Max(1, c.Column - 2)
    End If
    On Error GoTo 0
End Sub

' --- XUẤT KẾT QUẢ RA SHEET ---
Private Sub btnExport_Click()
    Dim outWs As Worksheet, i As Long, r As Long, wb As Workbook
    If lstResults.ListCount = 0 Then
        MsgBox "Không có dữ liệu kết quả để xuất!", vbExclamation, "Thông báo"
        Exit Sub
    End If
    
    Set wb = ActiveWorkbook
    On Error Resume Next
    Application.DisplayAlerts = False
    Set outWs = wb.Worksheets("KQ_TimKiem")
    If Not outWs Is Nothing Then outWs.Delete
    Application.DisplayAlerts = True
    On Error GoTo 0
    
    Set outWs = wb.Worksheets.Add(After:=wb.Sheets(wb.Sheets.Count))
    outWs.Name = "KQ_TimKiem"
    outWs.Tab.Color = RGB(0, 112, 192)
    
    outWs.Range("A1:E1").Merge
    outWs.Range("A1").Value = "KẾT QUẢ TÌM KIẾM TỪ KHÓA: """ & UCase(txtKeyword.Text) & """"
    outWs.Range("A1").Font.Bold = True: .Range("A1").Font.Color = RGB(255, 255, 255)
    outWs.Range("A1").Interior.Color = RGB(31, 78, 120): .Range("A1").RowHeight = 25
    outWs.Range("A1").VerticalAlignment = xlCenter
    
    outWs.Range("A2:E2").Value = Array("STT", "Tên Sheet", "Tọa độ ô", "Nội dung tìm thấy", "Truy cập nhanh")
    With outWs.Range("A2:E2")
        .Font.Bold = True: .Font.Color = RGB(255, 255, 255)
        .Interior.Color = RGB(46, 117, 182): .HorizontalAlignment = xlCenter: .RowHeight = 22
    End With
    
    For i = 0 To lstResults.ListCount - 1
        r = i + 3
        outWs.Cells(r, 1).Value = lstResults.List(i, 0)
        outWs.Cells(r, 2).Value = lstResults.List(i, 1)
        outWs.Cells(r, 3).Value = lstResults.List(i, 2)
        outWs.Cells(r, 4).Value = lstResults.List(i, 3)
        outWs.Hyperlinks.Add Anchor:=outWs.Cells(r, 5), Address:="", _
                             SubAddress:="'" & lstResults.List(i, 1) & "'!" & lstResults.List(i, 2), _
                             TextToDisplay:=">> Nhảy tới ô " & lstResults.List(i, 2)
    Next i
    
    outWs.Range("A2:E" & (lstResults.ListCount + 2)).Borders.LineStyle = xlContinuous
    outWs.Columns("A:E").AutoFit
    MsgBox "Đã xuất xong " & lstResults.ListCount & " kết quả sang sheet 'KQ_TimKiem'!", vbInformation, "Thành công"
End Sub
